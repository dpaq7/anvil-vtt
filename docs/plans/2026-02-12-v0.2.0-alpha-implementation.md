# v0.2.0 Alpha Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Transform Anvil v2 from a Director-only tool into a full multiplayer VTT with player agency, Draw Steel initiative, ability resolution, interactive non-battle scenes, audio playback, and demo-quality polish.

**Architecture:** Extend the existing WebSocket protocol with new message types for each feature. Server-side ability resolution using existing Logic modules. Client-side fog filtering. Side-based initiative replacing individual turn order. Toast notifications via Sonner for error feedback.

**Tech Stack:** React 19, TypeScript strict, PixiJS v8, Zustand, Cloudflare Workers + Durable Objects, Vitest, shadcn/ui + Sonner

static getFirstSide(roll: number): 'heroes' | 'villains' {
  return roll >= 6 ? 'heroes' : 'villains';
}

static isSideComplete(sideEntities: string[], actedThisRound: string[]): boolean {
  return sideEntities.every((id) => actedThisRound.includes(id));
}

static getOppositeSide(side: 'heroes' | 'villains'): 'heroes' | 'villains' {
  return side === 'heroes' ? 'villains' : 'heroes';
}

static isRoundOver(
  heroEntities: string[],
  villainEntities: string[],
  actedThisRound: string[],
): boolean {
  return (
    this.isSideComplete(heroEntities, actedThisRound) &&
    this.isSideComplete(villainEntities, actedThisRound)
  );
}

static createTurnActionState(speed: number): TurnActionState {
  return {
    mainActionUsed: false,
    maneuverUsed: false,
    moveRemaining: speed,
    triggeredUsedThisRound: false,
    mainConvertedTo: null,
  };
}

```typescript
private handleCombatAction(action: CombatAction, ws: WebSocket): void {
  switch (action.type) {
    case 'START_COMBAT': {
      const roll = this.rollD10(1)[0];
      const firstSide = roll >= 6 ? 'heroes' : 'villains';
      const combat: CombatState = {
        round: 1,
        activeSide: firstSide,
        firstSide,
        initiativeRoll: roll,
        heroEntities: action.heroEntityIds,
        villainEntities: action.villainEntityIds,
        actedThisRound: [],
        activeEntityId: null,
        malice: 0,
        turnActions: {},
      };
      this.sessionState.combat = combat;
      break;
    }
    case 'END_COMBAT': {
      this.sessionState.combat = null;
      break;
    }
    case 'CLAIM_TURN': {
      const combat = this.sessionState.combat;
      if (!combat) return;
      if (combat.activeSide !== 'heroes') return;
      if (!combat.heroEntities.includes(action.entityId)) return;
      if (combat.actedThisRound.includes(action.entityId)) return;
      if (combat.activeEntityId) return; // Someone already has a turn
      const entity = this.sessionState.entities.find((e) => e.id === action.entityId);
      const speed = typeof entity?.speed === 'number' ? entity.speed : 5;
      combat.activeEntityId = action.entityId;
      combat.turnActions[action.entityId] = {
        mainActionUsed: false,
        maneuverUsed: false,
        moveRemaining: speed,
        triggeredUsedThisRound: false,
        mainConvertedTo: null,
      };
      break;
    }
    case 'SELECT_TURN': {
      const combat = this.sessionState.combat;
      if (!combat) return;
      if (combat.activeSide !== 'villains') return;
      if (!combat.villainEntities.includes(action.entityId)) return;
      if (combat.actedThisRound.includes(action.entityId)) return;
      if (combat.activeEntityId) return;
      combat.activeEntityId = action.entityId;
      combat.turnActions[action.entityId] = {
        mainActionUsed: false,
        maneuverUsed: false,
        moveRemaining: 5,
        triggeredUsedThisRound: false,
        mainConvertedTo: null,
      };
      break;
    }
    case 'END_TURN': {
      const combat = this.sessionState.combat;
      if (!combat || !combat.activeEntityId) return;
      combat.actedThisRound.push(combat.activeEntityId);
      combat.activeEntityId = null;
      // Check if current side is complete
      const currentSideEntities =
        combat.activeSide === 'heroes' ? combat.heroEntities : combat.villainEntities;
      const sideComplete = currentSideEntities.every((id) =>
        combat.actedThisRound.includes(id),
      );
      if (sideComplete) {
        const otherSide = combat.activeSide === 'heroes' ? 'villains' : 'heroes';
        const otherSideEntities =
          otherSide === 'heroes' ? combat.heroEntities : combat.villainEntities;
        const otherSideComplete = otherSideEntities.every((id) =>
          combat.actedThisRound.includes(id),
        );
        if (otherSideComplete) {
          // Both sides complete — new round
          combat.round += 1;
          combat.actedThisRound = [];
          combat.activeSide = combat.firstSide;
          combat.malice += 2;
          combat.turnActions = {};
        } else {
          combat.activeSide = otherSide;
        }
      }
      break;
    }
    case 'ADJUST_MALICE': {
      const combat = this.sessionState.combat;
      if (!combat) return;
      combat.malice = Math.max(0, combat.malice + action.delta);
      break;
    }
    case 'APPLY_DAMAGE': {
      const entity = this.sessionState.entities.find((e) => e.id === action.entityId);
      if (!entity) return;
      const current = typeof entity.currentStamina === 'number' ? entity.currentStamina : 0;
      entity.currentStamina = current - action.amount;
      this.broadcast({ type: 'entity_updated', entityId: action.entityId, changes: { currentStamina: entity.currentStamina } });
      break;
    }
    case 'APPLY_HEALING': {
      const entity = this.sessionState.entities.find((e) => e.id === action.entityId);
      if (!entity) return;
      const current = typeof entity.currentStamina === 'number' ? entity.currentStamina : 0;
      const max = typeof entity.maxStamina === 'number' ? entity.maxStamina : current;
      entity.currentStamina = Math.min(max, current + action.amount);
      this.broadcast({ type: 'entity_updated', entityId: action.entityId, changes: { currentStamina: entity.currentStamina } });
      break;
    }
    case 'APPLY_CONDITION': {
      const entity = this.sessionState.entities.find((e) => e.id === action.entityId);
      if (!entity) return;
      const conditions = Array.isArray(entity.conditions) ? entity.conditions : [];
      if (!conditions.some((c: unknown) => typeof c === 'object' && c !== null && (c as Record<string, unknown>).id === action.condition)) {
        conditions.push({ id: action.condition, name: action.condition });
        entity.conditions = conditions;
      }
      this.broadcast({ type: 'entity_updated', entityId: action.entityId, changes: { conditions: entity.conditions } });
      break;
    }
    case 'REMOVE_CONDITION': {
      const entity = this.sessionState.entities.find((e) => e.id === action.entityId);
      if (!entity) return;
      const conditions = Array.isArray(entity.conditions) ? entity.conditions : [];
      entity.conditions = conditions.filter((c: unknown) => typeof c === 'object' && c !== null && (c as Record<string, unknown>).id !== action.conditionId);
      this.broadcast({ type: 'entity_updated', entityId: action.entityId, changes: { conditions: entity.conditions } });
      break;
    }
  }
  this.broadcast({ type: 'combat_updated', combat: this.sessionState.combat });
}
```

private handleNegotiationArgument(ws: WebSocket, skillId: string, approachText: string): void {
  if (!this.sessionState.negotiation) return;
  const meta = this.getMetaForSocket(ws);
  if (!meta) return;

  const dice = this.rollD10(2);
  const roll = dice[0] + dice[1]; // Simplified — no characteristic for now
  const tier = this.getTier(roll);
  const interestDelta = tier === 3 ? 1 : tier === 1 ? -1 : 0;

  this.sessionState.negotiation.interest = Math.max(0,
    Math.min(5, this.sessionState.negotiation.interest + interestDelta));

  const entry: ArgumentLogEntry = {
    id: crypto.randomUUID(),
    playerId: meta.userId,
    playerName: meta.username,
    skillId,
    approachText,
    roll,
    tier,
    interestDelta,
    timestamp: Date.now(),
  };
  this.sessionState.negotiation.argumentLog.push(entry);

  this.broadcast({
    type: 'negotiation_updated',
    interest: this.sessionState.negotiation.interest,
    patience: this.sessionState.negotiation.patience,
    argumentLog: this.sessionState.negotiation.argumentLog,
  });
}

private handleNegotiationPatience(delta: number): void {
  if (!this.sessionState.negotiation) return;
  this.sessionState.negotiation.patience = Math.max(0,
    this.sessionState.negotiation.patience + delta);

  this.broadcast({
    type: 'negotiation_updated',
    interest: this.sessionState.negotiation.interest,
    patience: this.sessionState.negotiation.patience,
    argumentLog: this.sessionState.negotiation.argumentLog,
  });
}

private handleMontageRoll(ws: WebSocket, skillId: string, characteristicId: string): void {
  if (!this.sessionState.montage) return;
  const meta = this.getMetaForSocket(ws);
  if (!meta) return;

  const dice = this.rollD10(2);
  const roll = dice[0] + dice[1];
  const tier = this.getTier(roll);
  const isSuccess = tier >= 2;

  if (isSuccess) {
    this.sessionState.montage.successes += 1;
  } else {
    this.sessionState.montage.failures += 1;
  }

  const entry: TestLogEntry = {
    id: crypto.randomUUID(),
    playerId: meta.userId,
    playerName: meta.username,
    skillId,
    characteristicId,
    roll,
    tier,
    outcome: isSuccess ? 'success' : 'failure',
    timestamp: Date.now(),
  };
  this.sessionState.montage.testLog.push(entry);

  // Check for completion
  if (this.sessionState.montage.successes >= this.sessionState.montage.successLimit) {
    this.sessionState.montage.outcome = 'success';
  } else if (this.sessionState.montage.failures >= this.sessionState.montage.failureLimit) {
    this.sessionState.montage.outcome = 'failure';
  }

  this.broadcast({
    type: 'montage_updated',
    successes: this.sessionState.montage.successes,
    failures: this.sessionState.montage.failures,
    testLog: this.sessionState.montage.testLog,
    outcome: this.sessionState.montage.outcome,
  });
}

private handleRespiteChoose(ws: WebSocket, activityId: string): void {
  if (!this.sessionState.respite) return;
  const meta = this.getMetaForSocket(ws);
  if (!meta) return;

  const activity = this.sessionState.respite.activities.find((a) => a.activityId === activityId);
  if (!activity || activity.claimedBy) return;

  activity.claimedBy = meta.userId;
  activity.claimedByName = meta.username;

  this.broadcast({
    type: 'respite_updated',
    activities: this.sessionState.respite.activities,
    completedBy: this.sessionState.respite.completedBy,
  });
}

private handleRespiteComplete(activityId: string): void {
  if (!this.sessionState.respite) return;
  const activity = this.sessionState.respite.activities.find((a) => a.activityId === activityId);
  if (!activity) return;

  activity.completed = true;
  if (activity.claimedBy) {
    const completedList = this.sessionState.respite.completedBy[activity.claimedBy] || [];
    completedList.push(activityId);
    this.sessionState.respite.completedBy[activity.claimedBy] = completedList;
  }

  this.broadcast({
    type: 'respite_updated',
    activities: this.sessionState.respite.activities,
    completedBy: this.sessionState.respite.completedBy,
  });
}

private handleAudioPlay(audioAssetId: string, loop: boolean): void {
  // For now, construct a simple URL — in production this would be an R2 presigned URL
  const audioUrl = `/api/assets/${audioAssetId}/data`;
  this.sessionState.audio = { playing: true, audioUrl, assetName: audioAssetId, loop };
  this.broadcast({ type: 'audio_command', action: 'play', audioUrl, assetName: audioAssetId, loop });
}

private handleAudioPause(): void {
  if (this.sessionState.audio) this.sessionState.audio.playing = false;
  this.broadcast({ type: 'audio_command', action: 'pause' });
}

private handleAudioStop(): void {
  this.sessionState.audio = null;
  this.broadcast({ type: 'audio_command', action: 'stop' });
}

private handleStoryUpdate(readAloudText: string): void {
  // Store in active scene data
  const sceneData = this.getActiveSceneData();
  if (sceneData) {
    sceneData['readAloudText'] = readAloudText;
  }
  this.broadcast({ type: 'story_updated', readAloudText });
}
